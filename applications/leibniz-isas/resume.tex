\section{Research Software}

\cvitem{helicopter}{
  \textbf{Multi-Modal Fluorescence Microscopy Image Analysis.}
  Resolves structural ambiguity in fluorescence, brightfield, phase-contrast, and
  spectral imaging data through sequential constraint exclusion across twelve independent
  measurement modalities. Reduces the configuration space from ${\sim}10^{60}$ possible
  assignments to a unique structural solution; achieves effective sub-nanometre
  resolution beyond any single optical channel. Directly applicable to 3D immune cell
  infiltration analysis in murine tissue.
  \href{https://github.com/fullscreen-triangle/helicopter}{github.com/fullscreen-triangle/helicopter}
}

\cvitem{lavoisier}{
  \textbf{MALDI MSI and LC-MS Analysis Framework.}
  Dual-pipeline architecture: numerical pipeline for MS1/MS2 ion annotation and
  feature extraction; visual pipeline converting spectra to temporal image sequences
  analysed by CNN (PyTorch). 98.9\,\% feature extraction accuracy against NIST
  reference libraries. Distributed computing via Ray; memory-mapped I/O for
  $>$100\,GB mzML datasets; Docker containerised.
  \href{https://github.com/fullscreen-triangle/lavoisier}{github.com/fullscreen-triangle/lavoisier}
}

\cvitem{gospel}{
  \textbf{Whole-Genome Variant Analysis and RNA-seq Integration.}
  Processes million-scale VCF files at $10^6$ variants/second; Bayesian network
  inference, Gaussian mixture models, variational Bayes. Validated against 1000 Genomes
  Phase 3, gnomAD v3.1.2, UK Biobank. Applicable to tumour immune-evasion genomics
  and somatic variant profiling.
  \href{https://github.com/fullscreen-triangle/gospel}{github.com/fullscreen-triangle/gospel}
}

\cvitem{syndrome}{
  \textbf{Cellular Disease State Computation.}
  Eight oscillator classes (protein, enzyme, channel, membrane, ATP, genetic expression,
  calcium, circadian); coherence index $\eta_i \in [0,1]$; disease vector
  $\mathbf{D}\in[0,1]^8$; $\mathcal{O}(N\log N)$ trajectory computation; sparse
  $\ell_1$ LP for therapeutic target selection. Applicable to cardiac and tumour
  microenvironment modelling.
  \href{https://github.com/fullscreen-triangle/syndrome}{github.com/fullscreen-triangle/syndrome}
}
