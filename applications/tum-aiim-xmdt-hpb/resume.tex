  \href{https://hieronymus-omega.vercel.app/}{hieronymus-omega.vercel.app}
}

  \textbf{Camera-based physiological inference.} Recovers melanin, haemoglobin, and

}
\cvitem{lavoisier}{

\cvitem{kwasa-kwasa, four-sided-triangle, purpose}{
