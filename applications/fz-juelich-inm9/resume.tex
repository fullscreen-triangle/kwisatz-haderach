\section{Research Software}

\cvitem{borgia}{
  \textbf{First-Principles Molecular Spectroscopy and Force Fields.}
  Derives electronic transitions, vibrational modes, absorption, and emission spectra
  from bounded phase space geometry --- zero empirical parameters.
  Triple Equivalence Theorem: oscillatory dynamics $=$ categorical state enumeration
  $=$ partition operations; GPU shader formally equivalent to physical spectrometer.
  Validated against NIST: nine elements, six molecules, self-consistency errors
  $<$1.63\,\%. Provides parameter-free baseline for molecular force contributions
  directly applicable to force field parameterisation.
  Live: \href{https://honjo-masamune.vercel.app/tools}{honjo-masamune.vercel.app/tools}.
  \href{https://github.com/fullscreen-triangle/borgia}{github.com/fullscreen-triangle/borgia}
}

\cvitem{shakespear}{
  \textbf{Protein Conformation and Trajectory Analysis.}
  Protein folding, trajectory, catalysis, and dynamics via Kuramoto oscillator
  synchronization in S-entropy coordinate space $(S_k, S_t, S_e)\in[0,1]^3$.
  Trajectory completion operator $T = P \circ B \circ K$ (Kirchhoff propagation,
  backward Viterbi MAP, thermodynamic projection) is a Banach contraction:
  unique fixed-point convergence to full conformational state.
  Complements MD by reconstructing trajectories backward from endpoint states ---
  applicable to folding pathway and transition-state characterisation.
  Browser-based, GPU-accelerated. Live:
  \href{https://shakespear-nine.vercel.app/instrument}{shakespear-nine.vercel.app/instrument}.
  \href{https://github.com/fullscreen-triangle/shakespear}{github.com/fullscreen-triangle/shakespear}
}

\cvitem{purpose + homo-veloce}{
  \textbf{Surrogate Models for Physics Solvers --- Backward Training Principle.}
  Formal proof that training at the pure-intent regime (hardest, fully-resolved
  instance) yields $\epsilon$-competence on all constrained sub-regimes by
  feasibility inclusion, not generalisation.
  Forward Training Collapse Theorem: provably positive VC failure gap for
  easy-to-hard curriculum; Sample Complexity Theorem gives $(N{+}1)\times$ savings.
  homo-veloce distils Gaussian process and GNN physics solvers into lightweight
  1D CNNs, LSTMs, Transformers for real-time inference --- the same pattern
  as learned interatomic potentials (NequIP, MACE, SchNet) for MD acceleration.
  \href{https://github.com/fullscreen-triangle/purpose}{github.com/fullscreen-triangle/purpose}
}

\cvitem{crown-prince}{
  \textbf{Geometric Pharmacology and Binding Affinity.}
  Dissociation constants $K_d$ derived from bounded phase space geometry;
  Hill equation as limiting case; PD/PK/ADME trajectories from partition structure.
  Validated: 39/39 NIST compounds. Applicable to protein--ligand binding
  free energy estimation without empirical docking parameters.
  \href{https://github.com/fullscreen-triangle/crown-prince}{github.com/fullscreen-triangle/crown-prince}
}

\cvitem{lavoisier}{

  98.9\,\% NIST accuracy; Ray distributed computing; $>$100\,GB dataset support;
